# Практична робота №5 — варіант 4

## Тема

Переривання, таймер і введення-виведення на x86.

## Мета

Навчитися виводити текст і зчитувати введення через переривання DOS, працювати з портами введення-виведення та встановлювати власний обробник переривання клавіатури з відновленням початкового вектора.

## Завдання варіанта

Визначити скан-код натиснутої клавіші `A` у шістнадцятковому форматі. Для клавіатурного набору Set 1 очікуваний make-код `A` — `1Eh`.

## Код програми

Програму написано для NASM у форматі DOS `.COM`, зібрано NASM 3.02 без помилок (розмір результату — 423 байти) та запущено в DOSBox-X. Повний код наведено нижче.

Програма демонструє виведення рядка через `int 21h` (`AH=09h`), введення символу через `AH=01h` і читання порту `60h`. Далі вона зберігає початковий вектор `int 9h` за допомогою `AH=35h`, встановлює власний обробник через `AH=25h`, розпізнає скан-коди клавіш `A` та `Esc` і відновлює початковий вектор перед завершенням.

```asm
bits 16
org 100h

jmp start

old_int9       dw 0, 0
a_scan_code    db 0
a_was_pressed  db 0
finished       db 0
input_char     db 0
port_value     db 0

hex_digits     db '0123456789ABCDEF'
title_msg      db 'Practical work 5, variant 4', 13, 10, '$'
input_msg      db 'Type a character: $'
port_msg       db 13, 10, 'Value read from port 60h: $'
wait_msg       db 13, 10, 'Press A, then press Esc to finish.', 13, 10, '$'
found_msg      db 13, 10, 'A scan code: $'
not_found_msg  db 13, 10, 'Key A was not pressed.', 13, 10, '$'
newline        db 13, 10, '$'

start:
    push cs
    pop ds

    mov dx, title_msg
    mov ah, 09h
    int 21h

    mov dx, input_msg
    mov ah, 09h
    int 21h
    mov ah, 01h
    int 21h
    mov [input_char], al

    mov dx, port_msg
    mov ah, 09h
    int 21h
    in al, 60h
    mov [port_value], al
    call print_hex_byte

    mov dx, newline
    mov ah, 09h
    int 21h

    ; Save the original INT 9h vector (offset:segment).
    mov ax, 3509h
    int 21h
    mov [old_int9], bx
    mov [old_int9 + 2], es

    mov byte [a_was_pressed], 0
    mov byte [finished], 0

    ; Install this program's INT 9h handler.
    mov dx, keyboard_isr
    mov ax, 2509h
    int 21h

    mov dx, wait_msg
    mov ah, 09h
    int 21h

wait_for_escape:
    cmp byte [finished], 1
    jne wait_for_escape

    ; Restore the original vector before returning to DOS.
    mov dx, [old_int9]
    mov ax, [old_int9 + 2]
    push ds
    mov ds, ax
    mov ax, 2509h
    int 21h
    pop ds

    cmp byte [a_was_pressed], 1
    jne key_a_not_found

    mov dx, found_msg
    mov ah, 09h
    int 21h
    mov al, [a_scan_code]
    call print_hex_byte
    jmp exit_program

key_a_not_found:
    mov dx, not_found_msg
    mov ah, 09h
    int 21h

exit_program:
    mov ax, 4C00h
    int 21h

; Print AL as two hexadecimal digits using DOS function AH=02h.
print_hex_byte:
    push ax
    push bx
    push cx

    mov bl, al
    mov al, bl
    mov cl, 4
    shr al, cl
    call print_hex_digit

    mov al, bl
    and al, 0Fh
    call print_hex_digit

    pop cx
    pop bx
    pop ax
    ret

print_hex_digit:
    push bx
    xor bh, bh
    mov bl, al
    mov dl, [hex_digits + bx]
    mov ah, 02h
    int 21h
    pop bx
    ret

; Read keyboard scan codes and acknowledge the keyboard controller and PIC.
keyboard_isr:
    push ax
    push ds
    push cs
    pop ds

    in al, 60h
    test al, 80h
    jnz acknowledge_keyboard

    cmp al, 1Eh
    jne check_escape
    mov [a_scan_code], al
    mov byte [cs:a_was_pressed], 1

check_escape:
    cmp al, 01h
    jne acknowledge_keyboard
    mov byte [cs:finished], 1

acknowledge_keyboard:
    ; Pulse bit 7 of port 61h to acknowledge the keyboard controller.
    in al, 61h
    mov ah, al
    or al, 80h
    out 61h, al
    mov al, ah
    out 61h, al

    mov al, 20h
    out 20h, al

    pop ds
    pop ax
    iret
```

## Пояснення роботи

1. `int 21h` з `AH=09h` виводить рядок, завершений символом `$`; `AH=01h` зчитує та відображає введений символ.
2. Команда `in al, 60h` читає байт з порту клавіатурного контролера. Виведене значення залежить від емулятора та стану контролера; це демонстрація прямого доступу до порту.
3. DOS-функція `AH=35h`, `AL=09h` повертає початковий вектор `int 9h` у `ES:BX`. Програма зберігає адресу зміщення й сегмент у змінній `old_int9`.
4. DOS-функція `AH=25h`, `AL=09h` встановлює власний обробник за адресою `DS:DX`. Обробник читає скан-код із порту `60h`: `1Eh` відповідає `A`, а `01h` — `Esc`. Для підтвердження клавіатури він формує імпульс через порт `61h` і надсилає EOI контролеру переривань через порт `20h`.
5. Після `Esc` початковий вектор відновлюється через `AH=25h` до завершення програми функцією `AH=4Ch`, щоб таблиця переривань не залишилася з адресою обробника, який уже не використовується.

## Результат запуску

Програму запущено в DOSBox-X. Після натискання `A`, а потім `Esc`, вона вивела скан-код `1E` і повернулася до командного рядка:

```text
Practical work 5, variant 4
Type a character:
Value read from port 60h: 39

Press A, then press Esc to finish.
A scan code: 1E
C:\>
```

Отриманий скан-код збігається з очікуваним для варіанта 4.

## Висновок

У роботі реалізовано власний обробник апаратного переривання клавіатури `int 9h`, який визначає make-код клавіші `A` (`1Eh`). Початковий вектор збережено перед установленням обробника та відновлено перед виходом, а правильність результату перевірено запуском у DOSBox-X.
